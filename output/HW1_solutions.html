def empirical_risk(X, y, theta):
    """Compute squared-loss empirical risk: sum((prediction - target)^2) / (2n)."""
    return np.sum((X @ theta - y) ** 2) / (2 * len(y))


def design_matrix(x, degree=1):
    """Build polynomial features [x, x^2, ..., x^degree, 1], including an intercept."""
    return np.column_stack([x ** k for k in range(1, degree + 1)] + [np.ones_like(x)])


def PolyRegress(x, y, d):
    """Fit a degree-d polynomial by least squares; return weights and training risk."""
    X_poly = design_matrix(x, d)
    theta_poly = np.linalg.lstsq(X_poly, y, rcond=None)[0]
    return theta_poly, empirical_risk(X_poly, y, theta_poly)

Q1. Linear Classification — Perceptron¶

Q1(a) One epoch and Q1(b) five epochs¶

Labels are $y\in\{-1,+1\}$. Initialize $\theta=0$ and $b=0$, then process the training examples in their original order. If $y(\theta^Tx+b)\leq0$, update $\theta\leftarrow\theta+yx$ and $b\leftarrow b+y$. Run the algorithm for one and five epochs without shuffling the data.

train = np.loadtxt(ROOT / '1/train_1_5.csv', delimiter=',')
test = np.loadtxt(ROOT / '1/test_1_5.csv', delimiter=',')


def perceptron(train, test, epochs):
    """Run sequential perceptron updates with zero-initialized weights and offset."""
    theta = np.zeros(train.shape[1] - 1)
    offset = 0.0
    for _ in range(epochs):
        for row in train:  # Keep the original order; the assignment forbids shuffling.
            x, y = row[:-1], row[-1]
            # A non-positive signed score means misclassification or a point on the boundary.
            if y * (theta @ x + offset) <= 0:
                theta += y * x
                offset += y
    pred = np.where(test[:, :-1] @ theta + offset >= 0, 1, -1)
    return theta, offset, np.mean(pred == test[:, -1])


for epochs in (1, 5):
    theta, offset, accuracy = perceptron(train, test, epochs)
    print(f'{epochs} epoch(s): theta={theta}, offset={offset:.1f}, test accuracy={accuracy:.4%}')

Q2(b) Batch Gradient Descent and Stochastic Gradient Descent¶

For squared loss $R(\theta)=\frac{1}{2n}\sum_i(x_i^T\theta-y_i)^2$, the gradient is $X^T(X\theta-y)/n$. Batch GD updates once per epoch using all examples; SGD updates after each example in the original data order. The learning rate is $\eta=0.01$, and each method runs for five epochs.

eta, n = 0.01, len(y)
# Batch GD uses the average gradient over all samples and updates once per epoch.
theta_gd = np.zeros(X.shape[1])
gd_risk = [empirical_risk(X, y, theta_gd)]
for _ in range(5):
    gradient = X.T @ (X @ theta_gd - y) / n

Q4. Support Vector Machines¶

Q4(a) Maximum-margin separator¶

A linear classifier uses the hyperplane $w^Tx+b=0$ as its decision boundary. Among hyperplanes that separate the training examples, an SVM selects one that maximizes the distance to the nearest examples. Under canonical scaling, the margin width is $2/\lVert w\rVert$, so maximizing the margin is equivalent to minimizing $\frac12\lVert w\rVert^2$. A large margin is generally more robust to small perturbations. Minimizing the margin instead would produce a narrow, fragile boundary that is sensitive to noise and can generalize poorly.

Q4(b) Soft margin and the parameter $C$¶

$$\min_{w,b,\xi}\;\frac12\lVert w\rVert^2+C\sum_i\xi_i\quad\text{s.t.}\quad y_i(w^T\phi(x_i)+b)\geq1-\xi_i,\;\xi_i\geq0.$$ Slack variables $\xi_i$ quantify margin violations. A small $C$ tolerates more violations in exchange for a wider margin; a large $C$ emphasizes training-set correctness and may overfit. Select $C$ using a validation set or cross-validation.

Q4(c) Kernel construction and verification¶

$K_1(x,x')=x^Tx'$ is the linear kernel, and $K_2(x,x')=(x^Tx')^2$ is a valid degree-two kernel. The sum and product of valid kernels are valid, so: $$K_+(x,x')=K_1+K_2=x^Tx'+(x^Tx')^2,\qquad K_\times(x,x')=K_1K_2=(x^Tx')^3.$$ The proposed $K(x,x')=\lVert x\rVert^2+\lVert x'\rVert^2-2x^Tx'=\lVert x-x'\rVert^2$ is not a valid kernel in general. For one-dimensional inputs 0 and 1, its Gram matrix is $\begin{bmatrix}0&1\\1&0\end{bmatrix}$, with eigenvalues 1 and -1. Since it is not positive semidefinite, the function fails the kernel condition.

Q5. Logistic Regression¶

Q5(a) Interpreting the predicted probability¶

Logistic regression outputs $h_\theta(x)=P(y=1\mid x;\theta)$. If $h_\theta(x)=0.65$, then $P(y=1\mid x;\theta)=0.65$ and $P(y=0\mid x;\theta)=1-0.65=0.35$. Therefore, the correct choices are 1 and 4.

import pandas as pd

train5 = pd.read_csv(ROOT / '5/train.csv')
test5 = pd.read_csv(ROOT / '5/test.csv')
feature_names = ['mean_radius', 'mean_texture']
X_train_raw = train5[feature_names].to_numpy(dtype=float)
y_train = train5['label'].to_numpy(dtype=float)
X_test_raw = test5[feature_names].to_numpy(dtype=float)
y_test = test5['label'].to_numpy(dtype=int)

# Fit scaling statistics on the training set only to prevent test-data leakage.
feature_mean = X_train_raw.mean(axis=0)
feature_std = X_train_raw.std(axis=0)
X_train_scaled = (X_train_raw - feature_mean) / feature_std
X_test_scaled = (X_test_raw - feature_mean) / feature_std
X_train_logistic = np.column_stack([np.ones(len(X_train_scaled)), X_train_scaled])
X_test_logistic = np.column_stack([np.ones(len(X_test_scaled)), X_test_scaled])


def sigmoid(z):
    """Map logits to probabilities; clipping prevents exponential overflow."""
    z = np.clip(z, -40, 40)
    return 1 / (1 + np.exp(-z))


def binary_cross_entropy(X, labels, weights):
    """Compute the mean binary cross-entropy using a numerically stable formula."""
    logits = X @ weights
    return float(np.mean(np.logaddexp(0, logits) - labels * logits))


weights = np.zeros(X_train_logistic.shape[1])
learning_rate = 0.1
epochs = 5000
train_losses = []
for _ in range(epochs):
    probabilities = sigmoid(X_train_logistic @ weights)
    # For logistic cross-entropy, the batch gradient is X.T @ (p - y) / n.
    gradient = X_train_logistic.T @ (probabilities - y_train) / len(y_train)
    weights -= learning_rate * gradient
    train_losses.append(binary_cross_entropy(X_train_logistic, y_train, weights))

test_probabilities = sigmoid(X_test_logistic @ weights)
test_predictions = (test_probabilities >= 0.5).astype(int)
test_accuracy = np.mean(test_predictions == y_test)
print(f'Train samples: {len(y_train)}, test samples: {len(y_test)}')
print('weights [intercept, standardized radius, standardized texture]:', weights)
print(f'Final training cross-entropy: {train_losses[-1]:.6f}')
print(f'Test accuracy: {test_accuracy:.4%}')

fig, axes = plt.subplots(1, 2, figsize=(12, 4.5))
axes[0].plot(np.arange(1, epochs + 1), train_losses, color='navy')
axes[0].set(xlabel='epoch', ylabel='binary cross-entropy', title='Logistic regression training loss')

radius_grid = np.linspace(X_train_raw[:, 0].min() - 1, X_train_raw[:, 0].max() + 1, 250)
texture_grid = np.linspace(X_train_raw[:, 1].min() - 1, X_train_raw[:, 1].max() + 1, 250)
RR, TT = np.meshgrid(radius_grid, texture_grid)
raw_grid = np.column_stack([RR.ravel(), TT.ravel()])
scaled_grid = (raw_grid - feature_mean) / feature_std
logits_grid = np.column_stack([np.ones(len(scaled_grid)), scaled_grid]) @ weights
probability_grid = sigmoid(logits_grid).reshape(RR.shape)
for label, name, color in [(0, 'Benign (0)', 'tab:blue'), (1, 'Malignant (1)', 'tab:orange')]:

# The decision boundary is the contour where the estimated positive-class probability equals 0.5.
axes[1].contour(RR, TT, probability_grid, levels=[0.5], colors='black', linewidths=2)
